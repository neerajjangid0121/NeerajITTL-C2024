import random
def calculateRolledNumber(noOfFaces):
    rolledNumber=random.randint(1, noOfFaces)
    return rolledNumber


def main():
    noOfFaces=6
    needsToRoll=True
    while needsToRoll:
        needsToRoll=input("Ready to roll? Enter Q to Quit")
        if needsToRoll.lower() !="q":
            rolledNumber=calculateRolledNumber(noOfFaces)
            print("You have rolled a",rolledNumber)
        else:
            needsToRoll=False



main();

def checkGuess(guessedNumber):
    if guessedNumber.isdigit() and 1<= int(guessedNumber) <=100:
        return True
    else:
        return False

def main():
    targetNumber=random.randint(1,100)
    isGuessCorrect=False
    guessedNumber=input("Guess a number between 1 and 100:")
    guessCounter=0
    while not isGuessCorrect:
        if not checkGuess(guessedNumber):
            guessedNumber=input("I wont count this one Please enter a number between 1 to 100")
            continue
        else:
            guessCounter+=1
            guessCounter=int(guessCounter)

        if guessCounter<targetNumber:
            guessCounter=input("Too low. Guess again")
        elif guessCounter>targetNumber:
            guessCounter=input("Too High. Guess again")
        else:
            print("You guessed it in",guessCounter,"guesses!")
            guessCounter=True


main()
